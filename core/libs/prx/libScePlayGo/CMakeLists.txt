    add_test_executable(guest_playgo_tests ${LIBS_INCLUDE_DIR}/tests/GuestPlayGo.cpp)

    configure_windows_unwind(guest_playgo_tests)
    add_test(NAME guest_playgo COMMAND guest_playgo_tests)
    if(WIN32)
        set_tests_properties(guest_playgo PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:$<TARGET_FILE_DIR:libScePlayGo>")
    endif()
    add_test_executable(guest_playgo_chunk_table_tests ${LIBS_INCLUDE_DIR}/tests/GuestPlayGoChunkTable.cpp)
    target_include_directories(guest_playgo_chunk_table_tests PRIVATE ${LIBS_INCLUDE_DIR})
    target_link_libraries(guest_playgo_chunk_table_tests PRIVATE libScePlayGo libc)
    configure_windows_unwind(guest_playgo_chunk_table_tests)
    foreach(mode IN ITEMS table xml)
        add_test(NAME guest_playgo_chunk_table_${mode} COMMAND guest_playgo_chunk_table_tests ${mode})
        if(WIN32)
            set_tests_properties(guest_playgo_chunk_table_${mode} PROPERTIES ENVIRONMENT_MODIFICATION
                "PATH=path_list_prepend:$<TARGET_FILE_DIR:libScePlayGo>")
        endif()
    endforeach()
endif()
